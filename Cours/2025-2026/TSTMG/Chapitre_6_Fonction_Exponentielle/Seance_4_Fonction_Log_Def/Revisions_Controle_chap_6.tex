\begin{questions}
\question Soit $a$ un nombre réel strictement positif. Simplifier les expressions suivantes :
\begin{parts}
\part $(a^4)^4 - a^2 \times a^4$
\part $(a^4)^4 \times (a^{-4})^2$
\part $\dfrac{(a^{-3})^{-2}}{a^4}$
\part $a^{-4} \times (a^3+a^{-5})$
\end{parts}
\question Pour chacune des fonctions suivantes, décrire ses variations.
\begin{parts}
\part $x \mapsto 10 \times 1,5^x$
\part $x \mapsto -13 \times 0,2^x$
\part $x \mapsto 7,7 \times 0,8^x$
\part $x \mapsto 0,8 \times 1,15^x$
\end{parts}
\question Dans les situations suivantes, calculer le taux d'évolution moyen.
\begin{parts}
\part Au bout de cinq ans, la population d'un pays a augmenté de $67\%$. Quelle est l'augmentation moyenne par année ?
\part On chauffe l'extrémité gauche d'une barre de fer longue de \qty{1}{\meter}. On constate que la température de l'extrémité droite a diminué de $58\%$. En déduire le taux d'évolution moyen par centimètre de la barre de fer. 
\end{parts}
\end{questions}
